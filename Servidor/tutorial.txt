# SERVER.JS

const express = require("express")
const mongoose = require("mongoose")
const bodyParser = require("body-parser")
const methodOverride = require("method-override")

const app = express()

const Quarto = require("./models/Quarto")

# CONECTAR MONGODB

mongoose.connect("mongodb://127.0.0.1:27017/hotel")

# CONFIGURAR EJS

app.set("view engine", "ejs")

# LER FORMULÁRIO

app.use(bodyParser.urlencoded({ extended: true }))

# ATIVAR PUT E DELETE

app.use(methodOverride("_method"))

# LIBERAR CSS

app.use(express.static("public"))

# ROTA PRINCIPAL

app.get("/", (req, res) => {

```
res.redirect("/quartos")
```

})

# ABRIR PÁGINA CADASTRO

app.get("/cadastro", (req, res) => {

```
res.render("cadastrar")
```

})

# CREATE

app.post("/quartos", async (req, res) => {

```
await Quarto.create({

    numero: req.body.numero,
    tipo: req.body.tipo,
    preco: req.body.preco,
    quantidade: req.body.quantidade

})

res.redirect("/quartos")
```

})

# READ

app.get("/quartos", async (req, res) => {

```
const quartos = await Quarto.find()

res.render("quartos", { quartos })
```

})

# ABRIR TELA EDITAR

app.get("/editar/:id", async (req, res) => {

```
const quarto = await Quarto.findById(req.params.id)

res.render("editar", { quarto })
```

})

# UPDATE

app.put("/quartos/:id", async (req, res) => {

```
await Quarto.findByIdAndUpdate(req.params.id, {

    numero: req.body.numero,
    tipo: req.body.tipo,
    preco: req.body.preco,
    quantidade: req.body.quantidade

})

res.redirect("/quartos")
```

})

# DELETE

app.delete("/quartos/:id", async (req, res) => {

```
await Quarto.findByIdAndDelete(req.params.id)

res.redirect("/quartos")
```

})

# RESERVAR QUARTO

app.post("/reservar/:id", async (req, res) => {

```
const quarto = await Quarto.findById(req.params.id)

if(quarto.quantidade > 0){

    quarto.quantidade--

    await quarto.save()

}

res.redirect("/quartos")
```

})

# LIGAR SERVIDOR

app.listen(80, () => {

```
console.log("Servidor rodando")
```

})

# MODEL QUARTO

const mongoose = require("mongoose")

```

tipo: String,
preco: Number,
quantidade: Number
```

})

module.exports = mongoose.model("Quarto", QuartoSchema)
